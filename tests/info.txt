test('Validate Status dropdown values', async ({ page }) => {

    const expectedStatus = [
        'Pending',
        'Generated',
        'Approved',
        'Uploaded',
        'Accepted',
        'Rejected',
        'Cancelled'
    ];

    // Click Bank dropdown
    await page.locator('input[aria-label="Bank"]').click();

    // Select first bank option
    const firstBank = page.locator('//div[@role="option"]').first();

    const bankName = await firstBank.textContent();

    console.log('Selected Bank : ', bankName);

    await firstBank.click();

    // Click Status dropdown
    await page.locator('input[aria-label="Status"]').click();

    // Validate status values
    for (const status of expectedStatus) {

        await expect(
            page.locator(`//div[@role="option" and normalize-space()="${status}"]`)
        ).toBeVisible();
    }
});
