import { Page, expect } from '@playwright/test';

export default class ConfigureEmployerPaycode {
  page: Page;

  constructor(page: Page) {
    this.page = page;
  }

  tabPayCodes = "//button[contains(.,'PAY CODES')]";
  tabPreTax = "//button[contains(.,'PRE TAX')]";
  tabPostTax = "//button[contains(.,'POST TAX')]";
  tabPercentage = "//button[contains(.,'PERCENTAGE')]";

  drpPaygroup = "(//input[@readonly])[1]";
  drpTransactionType = "(//input[@readonly])[2]";

  btnAdd = "//button[contains(.,'ADD')]";
  btnSave = "//button[contains(.,'SAVE')]";

  txtPayCode = "//input[contains(@id,'PayCode')]";
  txtDescription = "//input[contains(@id,'Description')]";
  drpDeductionType = "//input[contains(@id,'DeductionType')]";
  chkDefault = "//input[@type='checkbox']";

  async selectPaygroup(payGroup: string) {
    await this.page.locator(this.drpPaygroup).click();
    await this.page.getByText(payGroup, { exact: true }).click();
  }

  async selectTransactionType(transactionType: string) {
    await this.page.locator(this.drpTransactionType).click();
    await this.page.getByText(transactionType, { exact: true }).click();
  }

  async validateTabs() {
    await expect(this.page.locator(this.tabPayCodes)).toBeVisible();
    await expect(this.page.locator(this.tabPreTax)).toBeVisible();
    await expect(this.page.locator(this.tabPostTax)).toBeVisible();
    await expect(this.page.locator(this.tabPercentage)).toBeVisible();
  }

  async clickAddButton() {
    await this.page.locator(this.tabPayCodes).click();
    await this.page.locator(this.btnAdd).click();
  }

  async enterPaycodeDetails(TestData: any) {
    await this.page.locator(this.txtPayCode).fill(TestData.payCode);
    await this.page.locator(this.txtDescription).fill(TestData.description);

    await this.page.locator(this.drpDeductionType).click();
    await this.page.getByText(TestData.deductionType, { exact: true }).click();

    if (TestData.default === true) {
      await this.page.locator(this.chkDefault).first().check();
    }
  }

  async clickSave() {
    await this.page.locator(this.btnSave).click();
  }
}
