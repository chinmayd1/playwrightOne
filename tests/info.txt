async validateConsolidateAndRollbackButtonsFunctionality() {

    // Load batches and wait for API
    const [getBatchResponse] = await Promise.all([
        this.page.waitForResponse(response =>
            response.url().includes('/GetBatchData') &&
            response.status() === 200
        ),
        this.button_LoadBatches.click()
    ]);

    const batchData = await getBatchResponse.json();

    // If no records available
    if (batchData.length === 0) {

        console.log('No batch records available - Step Passed');
        return;
    }

    // Directly select first checkbox
    await this.checkbox_FirstRow.click();

    // Validate buttons enabled
    await expect(this.button_ConsolidateBatches).toBeEnabled();
    await expect(this.button_RollbackBatches).toBeEnabled();

    // =========================
    // Consolidate -> NO
    // =========================
    await this.button_ConsolidateBatches.click();

    await expect(
        this.popup_ApproveSelectedBatches
    ).toBeVisible();

    await this.button_No.click();

    // =========================
    // Rollback -> NO
    // =========================
    await this.button_RollbackBatches.click();

    await expect(
        this.popup_RollbackSelectedBatches
    ).toBeVisible();

    await this.button_No.click();

    // =========================
    // Consolidate -> YES
    // =========================
    await this.button_ConsolidateBatches.click();

    const [approveResponse] = await Promise.all([
        this.page.waitForResponse(response =>
            response.url().includes('/ApproveBatches') &&
            response.status() === 200
        ),
        this.button_Yes.click()
    ]);

    expect(approveResponse.status()).toBe(200);

    console.log('Approve API successful');

    // =========================
    // Reload and select again
    // =========================
    await Promise.all([
        this.page.waitForResponse(response =>
            response.url().includes('/GetBatchData') &&
            response.status() === 200
        ),
        this.button_LoadBatches.click()
    ]);

    await this.checkbox_FirstRow.click();

    // =========================
    // Rollback -> YES
    // =========================
    await this.button_RollbackBatches.click();

    const [rollbackResponse] = await Promise.all([
        this.page.waitForResponse(response =>
            response.url().includes('/RollbackBatches') &&
            response.status() === 200
        ),
        this.button_Yes.click()
    ]);

    expect(rollbackResponse.status()).toBe(200);

    console.log('Rollback API successful');
}
