async clickAddButton() {
  const addButton = this.page.locator('button.mud-button-root:has-text("Add")');

  await addButton.scrollIntoViewIfNeeded();
  await expect(addButton).toBeVisible();

  const box = await addButton.boundingBox();
  if (!box) throw new Error('Add button not found');

  await this.page.mouse.move(
    box.x + box.width / 2,
    box.y + box.height / 2
  );

  await this.page.mouse.down();
  await this.page.mouse.up();
}

async clickAddAndGetNewRow() {
  const rows = this.page.locator('tr.mud-table-row');
  const beforeCount = await rows.count();

  await this.clickAddButton();

  await expect(rows).toHaveCount(beforeCount + 1, {
    timeout: 10000
  });

  const newRow = rows.nth(beforeCount);
  await newRow.scrollIntoViewIfNeeded();

  return newRow;
}

async addPayCodeRow(
  payCode: string,
  deductionType: 'Pre Tax' | 'Post Tax' | 'Percentage'
) {
  const row = await this.clickAddAndGetNewRow();

  await row.locator('td[data-label="PayCode"] input').fill(payCode);
  await row.locator('td[data-label="Description"] input').fill(payCode);

  await row.locator('td[data-label="DeductionType"]').click({ force: true });

  await this.page
    .locator('.mud-popover-open, .mud-popover')
    .last()
    .getByText(deductionType, { exact: true })
    .click();
}
