readonly btnAdd = this.page.locator('button:has-text("Add")');

async clickAddAndGetNewRow() {
  const rows = this.page.locator('tr.mud-table-row');
  const beforeCount = await rows.count();

  await this.btnAdd.scrollIntoViewIfNeeded();
  await expect(this.btnAdd).toBeVisible();
  await expect(this.btnAdd).toBeEnabled();

  await this.btnAdd.click({ force: true });

  await expect(rows).toHaveCount(beforeCount + 1, {
    timeout: 10000
  });

  return rows.nth(beforeCount);
}

async clickAddAndGetNewRow() {
  const rows = this.page.locator('tr.mud-table-row');
  const beforeCount = await rows.count();

  const addButton = this.page.locator('button:has(span.mud-button-label:text("Add"))');

  await addButton.evaluate((btn: HTMLElement) => btn.click());

  await expect(rows).toHaveCount(beforeCount + 1, {
    timeout: 10000
  });

  return rows.nth(beforeCount);
}
