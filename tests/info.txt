async validateStatusDropdown(page) {

    const expectedStatus = [
        'Pending',
        'Generated',
        'Approved',
        'Uploaded',
        'Accepted',
        'Rejected',
        'Cancelled'
    ];

    // Click Bank dropdown
    await page.locator('input[aria-label="Bank"]').click();

    // Select first bank value
    const firstBankOption = page.locator('//div[@role="option"]').first();

    const bankName = await firstBankOption.textContent();

    console.log(`Selected Bank : ${bankName}`);

    await firstBankOption.click();

    // Click Status dropdown
    await page.locator('input[aria-label="Status"]').click();

    // Validate all status values
    for (const status of expectedStatus) {

        await expect(
            page.locator(`//div[@role="option" and normalize-space()="${status}"]`)
        ).toBeVisible();
    }

    console.log('All Status dropdown values are validated successfully');
}
