async validateStatusDropdown() {

  const expectedStatus = [
    'Pending',
    'Generated',
    'Approved',
    'Uploaded',
    'Accepted',
    'Rejected',
    'Cancelled'
  ];

  // Click Bank dropdown
  await $('input[aria-label="Bank"]').click();

  // Select first value from Bank dropdown
  const firstBankOption = await $$('//div[@role="option"]')[0];
  const bankName = await firstBankOption.getText();

  await firstBankOption.click();

  console.log(`Selected Bank: ${bankName}`);

  // Click Status dropdown
  await $('input[aria-label="Status"]').click();

  // Validate Status dropdown values
  for (const status of expectedStatus) {

    const statusOption = $(
      `//div[@role="option" and normalize-space()="${status}"]`
    );

    await expect(statusOption).toBeDisplayed();
  }
}
