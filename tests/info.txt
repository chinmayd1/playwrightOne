import { expect, Page } from '@playwright/test';

export class ConfigureEmployerPaycode {
  constructor(private page: Page) {}

  private btnSave = '//button[text()="SAVE" or text()="Save"]';
  private payCodeColumnCells = '//table//tbody//tr//td[1]';

  generatePayCode(
    deductionType: 'Pre Tax' | 'Post Tax' | 'Percentage',
    existingPayCodes: string[]
  ): string {
    let prefix: string;
    let maxNumber: number;

    switch (deductionType) {
      case 'Pre Tax':
        prefix = 'PRT';
        maxNumber = 9;
        break;

      case 'Post Tax':
        prefix = 'PT';
        maxNumber = 99;
        break;

      case 'Percentage':
        prefix = 'P';
        maxNumber = 999;
        break;
    }

    for (let i = 1; i <= maxNumber; i++) {
      const numberPart =
        deductionType === 'Pre Tax'
          ? i.toString()
          : deductionType === 'Post Tax'
            ? i.toString().padStart(2, '0')
            : i.toString().padStart(3, '0');

      const payCode = `${prefix}${numberPart}`;

      if (!existingPayCodes.includes(payCode)) {
        return payCode;
      }
    }

    throw new Error(`No unique paycode available for ${deductionType}`);
  }

  async getExistingPayCodesFromGrid(): Promise<string[]> {
    return await this.page
      .locator(this.payCodeColumnCells)
      .allTextContents();
  }

  async getUniquePayCode(
    deductionType: 'Pre Tax' | 'Post Tax' | 'Percentage'
  ): Promise<string> {
    const existingPayCodes = await this.getExistingPayCodesFromGrid();

    return this.generatePayCode(
      deductionType,
      existingPayCodes.map(x => x.trim())
    );
  }

  async validateSavePayCode(
    payCode: string,
    isDefaultExpected: boolean
  ): Promise<void> {
    const saveResponsePromise = this.page.waitForResponse(response =>
      response.url().includes('/api/SavePayCodes') &&
      response.request().method() === 'POST'
    );

    await this.page.locator(this.btnSave).click();

    const saveResponse = await saveResponsePromise;

    expect(saveResponse.status()).toBe(200);
    expect(saveResponse.ok()).toBeTruthy();

    const getResponse = await this.page.waitForResponse(response =>
      response.url().includes('/api/GetPayCodes') &&
      response.request().method() === 'GET'
    );

    expect(getResponse.status()).toBe(200);
    expect(getResponse.ok()).toBeTruthy();

    const responseBody = await getResponse.json();

    const addedPayCode = responseBody.find(
      (item: any) => item.payCode === payCode
    );

    expect(addedPayCode).toBeDefined();
    expect(addedPayCode.payCode).toBe(payCode);
    expect(addedPayCode.status).toBe('Active');

    const defaultPayCodes = responseBody.filter(
      (item: any) => item.isDefault === true
    );

    expect(defaultPayCodes.length).toBe(1);

    if (isDefaultExpected) {
      expect(addedPayCode.isDefault).toBe(true);
      expect(defaultPayCodes[0].payCode).toBe(payCode);
    }
  }
}

const deductionType = 'Percentage';

const payCode =

generateRandomPayCode(
  deductionType: 'Pre Tax' | 'Post Tax' | 'Percentage',
  existingPayCodes: string[] = []
): string {

  const chars = 'ABCDEFGHIJKLMNOPQRSTUVWXYZ0123456789';

  let prefix = '';

  switch (deductionType) {
    case 'Pre Tax':
      prefix = 'PR';
      break;

    case 'Post Tax':
      prefix = 'PT';
      break;

    case 'Percentage':
      prefix = 'P';
      break;
  }

  let payCode = '';

  do {
    const randomPartLength = 4 - prefix.length;

    const randomPart = Array.from(
      { length: randomPartLength },
      () => chars[Math.floor(Math.random() * chars.length)]
    ).join('');

    payCode = `${prefix}${randomPart}`;

  } while (existingPayCodes.includes(payCode));

  return payCode;
}

async enterPayCodeDetails(
  payCode: string,
  description: string,
  deductionType: 'Pre Tax' | 'Post Tax' | 'Percentage'
) {
  const lastRow = this.page.locator('tr.mud-table-row').last();

  await lastRow
    .locator('td[data-label="PayCode"] input')
    .fill(payCode);

  await lastRow
    .locator('td[data-label="Description"] input')
    .fill(description);

  await lastRow
    .locator('td[data-label="DeductionType"] .mud-select-input')
    .click();

  await this.page
    .locator('.mud-popover')
    .getByText(deductionType, { exact: true })
    .click();
}

await configureEmployerPaycode.enterPayCodeDetails(
  payCode,
  payCode,
  'Percentage'
);
