import { test, expect } from '@playwright/test';

async function validateBankDropdown(page) {

    // Reload page and wait for API response
    const [bankApiResponse] = await Promise.all([

        page.waitForResponse(response =>
            response.url().includes('/GetBankCodes') &&
            response.status() === 200
        ),

        page.reload()
    ]);

    // Convert API response to JSON
    const responseBody = await bankApiResponse.json();

    // Extract bank codes from API
    const apiBankValues = responseBody.map(bank =>
        bank.bankCode.trim()
    );

    console.log('API Values:', apiBankValues);

    // Click Bank dropdown input
    await page.locator('input[aria-label="Bank"]').click();

    // Wait for dropdown options to appear
    await page.waitForSelector('[role="option"]');

    // Capture UI dropdown values
    const uiValues = await page
        .locator('[role="option"]')
        .allTextContents();

    // Remove extra spaces
    const actualUiValues = uiValues.map(value =>
        value.trim()
    );

    console.log('UI Values:', actualUiValues);

    // Validate API vs UI
    expect(actualUiValues.sort())
        .toEqual(apiBankValues.sort());
}

test('Validate Bank Dropdown Values', async ({ page }) => {

    // Open page
    await page.goto(
        'https://salpac-uat01.test.smart.com.au/processpayments/submit'
    );

    // Validate dropdown values
    await validateBankDropdown(page);
});
