import { expect, Page, Locator } from '@playwright/test';

export class ConfigureEmployerPaycode {
  constructor(private page: Page) {}

  readonly btnAdd = this.page.getByRole('button', { name: 'ADD' });

  async generateRandomPayCode(
    deductionType: 'Pre Tax' | 'Post Tax' | 'Percentage'
  ): Promise<string> {

    const chars = 'ABCDEFGHIJKLMNOPQRSTUVWXYZ0123456789';

    let prefix = '';

    switch (deductionType) {
      case 'Pre Tax':
        prefix = 'PR';
        break;

      case 'Post Tax':
        prefix = 'PT';
        break;

      case 'Percentage':
        prefix = 'P';
        break;
    }

    const randomLength = 4 - prefix.length;

    const randomPart = Array.from(
      { length: randomLength },
      () => chars[Math.floor(Math.random() * chars.length)]
    ).join('');

    return `${prefix}${randomPart}`;
  }

  async clickAddAndGetLastRow(): Promise<Locator> {

    await this.btnAdd.click();

    const tableContainer = this.page.locator('.mud-table-container');

    await tableContainer.evaluate((el: any) => {
      el.scrollTop = el.scrollHeight;
    });

    await this.page.waitForTimeout(1000);

    const lastRow = this.page
      .locator('tr.mud-table-row')
      .last();

    await lastRow.scrollIntoViewIfNeeded();

    return lastRow;
  }

  async selectDeductionType(
    row: Locator,
    deductionType: 'Pre Tax' | 'Post Tax' | 'Percentage'
  ) {

    await row
      .locator('td[data-label="DeductionType"]')
      .click({ force: true });

    await this.page
      .locator('.mud-popover')
      .getByText(deductionType, { exact: true })
      .click();
  }

  async addPayCode(
    deductionType: 'Pre Tax' | 'Post Tax' | 'Percentage'
  ): Promise<string> {

    const payCode =
      await this.generateRandomPayCode(deductionType);

    const row = await this.clickAddAndGetLastRow();

    await row
      .locator('td[data-label="PayCode"] input')
      .fill(payCode);

    await row
      .locator('td[data-label="Description"] input')
      .fill(payCode);

    await this.selectDeductionType(
      row,
      deductionType
    );

    return payCode;
  }

  async validateSaveAndGetPayCode(
    payCode: string,
    isDefaultExpected: boolean
  ) {

    const savePromise = this.page.waitForResponse(
      response =>
        response.url().includes('/SavePayCodes') &&
        response.request().method() === 'POST'
    );

    await this.page
      .getByRole('button', { name: 'SAVE' })
      .click();

    const saveResponse = await savePromise;

    expect(saveResponse.status()).toBe(200);

    const getPromise = this.page.waitForResponse(
      response =>
        response.url().includes('/GetPayCodes') &&
        response.request().method() === 'GET'
    );

    const getResponse = await getPromise;

    expect(getResponse.status()).toBe(200);

    const responseBody =
      await getResponse.json();

    const addedPayCode = responseBody.find(
      (x: any) => x.payCode === payCode
    );

    expect(addedPayCode).toBeDefined();

    const defaultPayCodes = responseBody.filter(
      (x: any) => x.isDefault === true
    );

    expect(defaultPayCodes.length).toBe(1);

    if (isDefaultExpected) {
      expect(addedPayCode.isDefault).toBe(true);
      expect(defaultPayCodes[0].payCode)
        .toBe(payCode);
    }
  }
}

const payCode =
  await configureEmployerPaycode.addPayCode(
    'Percentage'
  );

await configureEmployerPaycode
  .validateSaveAndGetPayCode(
    payCode,
    true
  );
