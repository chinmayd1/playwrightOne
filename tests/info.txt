import { test, expect } from '@playwright/test';
import fs from 'fs';
import pdf from 'pdf-parse';

test('Validate PDF content', async ({ page }) => {

  const pdfResponsePromise = page.waitForResponse(response =>
    response.url().includes('/WebReport') &&
    response.status() === 200
  );

  await page.goto(
    'https://salpac-uat01.test.smart.com.au/packagemanagement/spas?employeeId=278958'
  );

  const response = await pdfResponsePromise;

  // Get PDF buffer
  const pdfBuffer = await response.body();

  // Save PDF (optional)
  fs.writeFileSync('spas-report.pdf', pdfBuffer);

  // Parse PDF
  const data = await pdf(pdfBuffer);

  console.log(data.text);

  // Validate content
  expect(data.text).toContain('Activity Statement');

  expect(data.text).toContain('ALEXIS ALEXIS');

  expect(data.text).toContain('Payroll Number');

});

const fs = require("fs");
const { test, expect } = require("@playwright/test");

test("Validate Activity Statement PDF", async ({ page }) => {

  // Wait for PDF API response
  const pdfResponsePromise = page.waitForResponse(resp =>
    resp.url().includes("GetReportDocument") &&
    resp.status() === 200
  );

  // Open report
  await PackagePage.hoverOnMainMenuAndClickSubMenu(
    "Reports",
    TestData.subMenu
  );

  // Get PDF response
  const pdfResponse = await pdfResponsePromise;

  // Validate response status
  expect(pdfResponse.status()).toBe(200);

  // Read PDF buffer
  const pdfBuffer = await pdfResponse.body();

  // Validate actual PDF file
  expect(pdfBuffer.slice(0, 5).toString()).toBe("%PDF-");

  // Validate PDF not empty/corrupt
  expect(pdfBuffer.length).toBeGreaterThan(1000);

  // Convert PDF binary into readable string
  const pdfContent = pdfBuffer.toString("latin1");

  console.log(pdfContent);

  // Validate content from binary
  expect(pdfContent).toContain("ALEXIS");
  expect(pdfContent).toContain("Activity");
  expect(pdfContent).toContain("Payroll");

  // Save PDF for debugging/evidence
  fs.writeFileSync("activity-statement.pdf", pdfBuffer);

  // Validate PDF viewer opened
  await expect(
    page.locator("iframe, canvas").first()
  ).toBeVisible();

  console.log("PDF validated successfully");
});

const fs = require("fs");
const { test, expect } = require("@playwright/test");

test("Validate Activity Statement using source API", async ({ page }) => {
  const jsonResponses = [];

  page.on("response", async (response) => {
    const contentType = response.headers()["content-type"] || "";

    if (
      response.status() === 200 &&
      contentType.includes("application/json")
    ) {
      try {
        const data = await response.json();

        jsonResponses.push({
          url: response.url(),
          data
        });
      } catch (e) {}
    }
  });

  const pdfResponsePromise = page.waitForResponse(resp =>
    resp.url().includes("GetReportDocument") &&
    resp.status() === 200
  );

  await PackagePage.hoverOnMainMenuAndClickSubMenu(
    "Reports",
    TestData.subMenu
  );

  const pdfResponse = await pdfResponsePromise;
  const pdfBuffer = await pdfResponse.body();

  expect(pdfBuffer.slice(0, 5).toString()).toBe("%PDF-");
  expect(pdfBuffer.length).toBeGreaterThan(1000);

  fs.writeFileSync("activity-statement.pdf", pdfBuffer);

  await page.waitForTimeout(3000);

  const matchedApi = jsonResponses.find(api =>
    JSON.stringify(api.data).toLowerCase().includes("alexis")
  );

  expect(matchedApi).toBeTruthy();

  console.log("Matched API URL:", matchedApi.url);
  console.log("Matched API Data:", JSON.stringify(matchedApi.data, null, 2));

  const sourceData = JSON.stringify(matchedApi.data);

  expect(sourceData).toMatch(/Alexis\s+Alexis/i);
  expect(sourceData).toMatch(/4004388/i);
  expect(sourceData).toMatch(/Royal\s+Children/i);
  expect(sourceData).toMatch(/01\s+Apr\s+2026/i);
  expect(sourceData).toMatch(/18\s+May\s+2026/i);

  await expect(page.locator("iframe, embed, canvas").first()).toBeVisible();
});
