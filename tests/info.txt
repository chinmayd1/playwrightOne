async validateStatusDropdown(page) {

    const expectedStatus = [
        'Pending',
        'Generated',
        'Approved',
        'Uploaded',
        'Accepted',
        'Rejected',
        'Cancelled'
    ];

    // Open Bank dropdown
    await page.locator('input[aria-label="Bank"]').click();

    // Wait for bank options
    await page.waitForSelector('[role="option"]');

    // Select first bank
    const firstBankOption = page.locator('[role="option"]').first();

    const bankName = await firstBankOption.textContent();

    console.log(`Selected Bank : ${bankName?.trim()}`);

    await firstBankOption.click();

    // Open Status dropdown
    await page.locator('input[aria-label="Status"]').click();

    // Wait for status dropdown options
    await page.waitForSelector('[role="option"]');

    // Capture all status dropdown values
    const actualStatusValues = await page
        .locator('[role="option"]')
        .allTextContents();

    const formattedStatusValues = actualStatusValues.map(status =>
        status.trim()
    );

    console.log('Actual Status Values :', formattedStatusValues);

    // Validate expected vs actual status values
    expect(formattedStatusValues.sort())
        .toEqual(expectedStatus.sort());

    console.log('All Status dropdown values are validated successfully');
}
