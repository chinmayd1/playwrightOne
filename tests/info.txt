test("Validate Activity Statement using source API", async ({ page }) => {
  const jsonResponses = [];

  page.on("response", async (response) => {
    const contentType = response.headers()["content-type"] || "";

    if (
      response.status() === 200 &&
      contentType.includes("application/json")
    ) {
      try {
        const data = await response.json();

        jsonResponses.push({
          url: response.url(),
          data
        });
      } catch (e) {}
    }
  });

  const pdfResponsePromise = page.waitForResponse(resp =>
    resp.url().includes("GetReportDocument") &&
    resp.status() === 200
  );

  await PackagePage.hoverOnMainMenuAndClickSubMenu(
    "Reports",
    TestData.subMenu
  );

  const pdfResponse = await pdfResponsePromise;
  const pdfBuffer = await pdfResponse.body();

  expect(pdfBuffer.slice(0, 5).toString()).toBe("%PDF-");
  expect(pdfBuffer.length).toBeGreaterThan(1000);

  fs.writeFileSync("activity-statement.pdf", pdfBuffer);

  await page.waitForTimeout(3000);

  const matchedApi = jsonResponses.find(api =>
    JSON.stringify(api.data).toLowerCase().includes("alexis")
  );

  expect(matchedApi).toBeTruthy();

  console.log("Matched API URL:", matchedApi.url);
  console.log("Matched API Data:", JSON.stringify(matchedApi.data, null, 2));

  const sourceData = JSON.stringify(matchedApi.data);

  expect(sourceData).toMatch(/Alexis\s+Alexis/i);
  expect(sourceData).toMatch(/4004388/i);
  expect(sourceData).toMatch(/Royal\s+Children/i);
  expect(sourceData).toMatch(/01\s+Apr\s+2026/i);
  expect(sourceData).toMatch(/18\s+May\s+2026/i);

  await expect(page.locator("iframe, embed, canvas").first()).toBeVisible();
});

      pdfBuffer.slice(0, 5).toString()

  async savePdf(pdfBuffer, fileName = "activity-statement.pdf") {

import { expect } from "@playwright/test";
import pdf from "pdf-parse";

export default class PacmanReportsPage {

  constructor(page) {
    this.page = page;
  }

  // ===============================================
  // WAIT FOR PDF RESPONSE
  // ===============================================

  async waitForPdfResponse() {

    return await this.page.waitForResponse(
      (resp) =>
        resp.url().includes("/WebReports/GetReportDocument") &&
        resp.status() === 200
    );

  }

  // ===============================================
  // CLICK REPORT SUBMENU
  // ===============================================

  async clickReportSubMenu(subMenu, PackagePage) {

    await PackagePage.hoverOnMainMenuAndClickSubMenu(
      "Reports",
      subMenu
    );

  }

  // ===============================================
  // GET PDF BUFFER
  // ===============================================

  async getPdfBuffer(pdfResponse) {

    return await pdfResponse.body();

  }

  // ===============================================
  // EXTRACT PDF TEXT
  // ===============================================

  async extractPdfText(pdfBuffer) {

    const pdfData = await pdf(pdfBuffer);

    return pdfData.text;

  }

  // ===============================================
  // COMPLETE COMMON PDF FLOW
  // ===============================================

  async getPdfText(testData, PackagePage) {

    const pdfResponsePromise =
      this.waitForPdfResponse();

    await this.clickReportSubMenu(
      testData.subMenu,
      PackagePage
    );

    const pdfResponse =
      await pdfResponsePromise;

    const pdfBuffer =
      await this.getPdfBuffer(pdfResponse);

    const pdfText =
      await this.extractPdfText(pdfBuffer);

    return {
      pdfResponse,
      pdfText
    };

  }

  // ===============================================
  // VERIFY CONTENT LOADED
  // ===============================================

  async verifyContentLoaded(
    page,
    MyReporter,
    pdfResponse,
    pdfText
  ) {

    try {

      expect(pdfResponse.status()).toBe(200);

      expect(pdfText.length)
        .toBeGreaterThan(1000);

      await MyReporter.report(
        page,
        "✓ Complete report content loaded"
      );

    } catch (error) {

      await MyReporter.report(
        page,
        "✗ Report content not loaded properly"
      );

      throw error;
    }

  }

  // ===============================================
  // GENERIC PDF TEXT VALIDATION
  // ===============================================

  async verifyPdfText(
    page,
    MyReporter,
    pdfText,
    expectedText,
    label
  ) {

    try {

      expect(pdfText).toMatch(
        new RegExp(expectedText, "i")
      );

      await MyReporter.report(
        page,
        `✓ ${label} verified`
      );

    } catch (error) {

      await MyReporter.report(
        page,
        `✗ ${label} not found`
      );

      throw error;
    }

  }

}

async validateBankFilterUsingApiResponse() {

    // Click button and wait for API response together
    const [response] = await Promise.all([
        this.page.waitForResponse(response =>
            response.url().includes('/GetBatchData') &&
            response.status() === 200
        ),
        this.button_LoadBatches.click()
    ]);

    const responseArray = await response.json();

    // If API response has records
    if (responseArray.length > 0) {

        // Pick first bank name
        const bankName = responseArray[0].bank;

        console.log(`Selected Bank Name: ${bankName}`);

        // Enter bank name in Bank filter textbox
        await this.page
            .locator("//input[@placeholder='Filter value']")
            .nth(0)
            .fill(bankName);

        // Wait for table filtering
        await this.page.waitForTimeout(2000);

        // Validate all visible rows contain same bank
        const bankColumnValues = this.page.locator("//table//tbody//tr//td[2]");

        const rowCount = await bankColumnValues.count();

        for (let i = 0; i < rowCount; i++) {
            await expect(bankColumnValues.nth(i)).toContainText(bankName);
        }

        console.log('Bank filter validation successful');

    } else {

        console.log('API response array is empty - Step Passed');

    }
}
